взяти визначення мавка;

простір мавка {
  внутрішня дія рідна_дія_методу_структури_Словник__значення(
    М: *Машина,
    діяльність: *Діяльність
  ) -> ЗначенняАбоСтанПадіння {
    якщо діяльність.здійснювач.вид != ЗПредмет {
      вернути ПАДІННЯ(М, створити_текст(М, "Неможливо здійснити дію словника \"значення\"."));
    }

    змінна предмет_словника = діяльність.здійснювач.дані як *Предмет як *ПредметСловника;

    якщо предмет_словника.клас != М.клас_Словник {
      вернути ПАДІННЯ(М, створити_текст(М, "Неможливо здійснити дію словника \"значення\"."));
    }

    змінна п: природне = 0;

    змінна елементи = виділити<Значення>(М, предмет_словника.розмір);

    поки п < предмет_словника.розмір {
      елементи[п] = предмет_словника.елементи[п].значення;

      п += 1;
    }

    вернути ПРЕДМЕТ(створити_послідовність_без_копіювання(
      М,
      предмет_словника.розмір,
      елементи
    ));
  }

  дія записати_метод_класу_Словник__значення(
    М: *Машина
  ) {
    змінна предмет_дії = створити_рідну_дію(
      М,
      М.константи_назв._значення,
      М.пуста_послідовність,
      пусто,
      рідна_дія_методу_структури_Словник__значення,
      пусто
    );

    записати_метод_класу(
      М,
      М.клас_Словник,
      М.константи_назв._значення,
      предмет_дії
    );

    якщо М.стан_падіння {
      ЗАПАНІКУВАТИ(М, "Помилка при записі дії \"значення\" для структури \"Словник\".");
    }
  }
}
