дія КД_виділити_сиру_памʼять(система: адреса<КД::Система>, розмір: природне) -> адреса<п8> {
  вернути пристрій_мавки_виділити_сиру_памʼять(розмір);
}

дія КД_перевиділити_сиру_памʼять(система: адреса<КД::Система>, значення: адреса<п8>, новий_розмір: природне) -> адреса<п8> {
  вернути пристрій_мавки_перевиділити_сиру_памʼять(значення, новий_розмір);
}

дія КД_звільнити_сиру_памʼять(система: адреса<КД::Система>, значення: адреса<ніщо>) {
  пристрій_мавки_звільнити_сиру_памʼять(значення як адреса<п8>);
}

дія пристрій_мавки_перекодувати_ю8_в_кд(розмір: природне, дані: адреса<п8>, вихід_розміру: адреса<природне>, вихід_даних: адреса<адреса<п8>>, вихід_позиції_помилки: адреса<природне>) -> логічне {
  змінна система_КД = КД::Система {
    дані = пусто,
    виділити_сиру_памʼять = КД_виділити_сиру_памʼять,
    звільнити_сиру_памʼять = КД_звільнити_сиру_памʼять,
    перевиділити_сиру_памʼять = КД_перевиділити_сиру_памʼять
  };

  змінна вихід: кд;
  змінна позиція_помилки: природне;

  змінна успіх = КД::перекодувати_з_ю8(
    система_КД::адреса,
    ю8 { розмір, дані },
    ні,
    ні,
    вихід::адреса,
    позиція_помилки::адреса
  );

  якщо успіх {
    вихід_розміру::вміст = вихід.розмір;
    вихід_даних::вміст = вихід.дані;
  }

  вернути успіх;
}
