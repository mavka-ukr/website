взяти визначення мавка/бібліотека;
взяти визначення мавка/пристрій;

простір мавка {
  простір бібліотека {
    внутрішня дія рідна_дія_методу_структури_Слуга__зупинити(
      М: *Машина,
      діяльність: *Діяльність
    ) -> ЗначенняАбоСтанПадіння {
      змінна кдбм = М.користувацькі_дані_бібліотеки.значення як *КористувацькіДаніБібліотекиМавки;

      якщо діяльність.здійснювач.вид != ЗПредмет {
        вернути ПАДІННЯ(М, створити_текст(М, "Неможливо здійснити метод \"зупинити\"."));
      }

      змінна предмет_слуги = діяльність.здійснювач.дані як *Предмет як *СкладенийПредмет;

      якщо предмет_слуги.клас != кдбм.інет__Слуга {
        вернути ПАДІННЯ(М, створити_текст(М, "Неможливо здійснити дію \"зупинити\"."));
      }

      змінна задіяне_закрити_підключення = НЕДІЙСНЕ;
      змінна знайдено_задіяне_закрити_підключення = ні;

      знайдено_задіяне_закрити_підключення = знайти_задіяне(М, діяльність.задіяні, 0, М.константи_назв._закрити_підключення, задіяне_закрити_підключення::адреса);

      змінна кдс = предмет_слуги.користувацькі_дані.значення як *КористувацькіДаніІнетСлуги;
      якщо кдс.адр_слуга == пусто {
        вернути НЕДІЙСНЕ;
      }

      бібліотека_мавки_інет_зупинити_слугу(
        кдс.адр_слуга,
        природне(ЧИ(задіяне_закрити_підключення)),
      );

      вернути ДІЙСНЕ;
    }

    внутрішня дія записати_метод_класу_Слуга__зупинити(
      М: *Машина,
      клас: *ПредметКласу
    ) {
      змінна предмет_дії = створити_рідну_дію(
        М,
        М.константи_назв._зупинити,
        М.пуста_послідовність,
        пусто,
        рідна_дія_методу_структури_Слуга__зупинити,
        пусто
      );

      записати_метод_класу(
        М,
        клас,
        М.константи_назв._зупинити,
        предмет_дії
      );

      якщо М.стан_падіння {
        ЗАПАНІКУВАТИ(М, "Помилка при записі методу \"зупинити\" для класу \"Слуга\".");
      }
    }

    дія записати_клас__інет__Слуга(
      М: *Машина,
      предмет_модуля: *ПредметМодуля
    ) {
      змінна кдбм = М.користувацькі_дані_бібліотеки.значення як *КористувацькіДаніБібліотекиМавки;

      кдбм.інет__Слуга = створити_клас(
        М,
        М.константи_назв._Слуга,
        М.пуста_послідовність
      );

      записати_метод_класу_Слуга__зупинити(М, кдбм.інет__Слуга);

      записати_властивість_модуля(
        М,
        предмет_модуля,
        М.константи_назв._Слуга,
        ПРЕДМЕТ(кдбм.інет__Слуга)
      );
    }
  }
}
