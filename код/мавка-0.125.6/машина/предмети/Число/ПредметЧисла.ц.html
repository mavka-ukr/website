взяти визначення мавка;

внутрішня дія копіювати_памʼять<Т>(
  куди: *Т,
  звідки: *Т,
  кількість: природне
) {
  змінна п: природне = 0;

  поки п < кількість {
    куди[п] = звідки[п];

    п += 1;
  }
}

внутрішня дія заповнити_памʼять<Т>(
  куди: *Т,
  значення: Т,
  кількість: природне
) {
  змінна п: природне = 0;

  поки п < кількість {
    куди[п] = значення;

    п += 1;
  }
}

      Місцезнаходження { пусто, 0 }
