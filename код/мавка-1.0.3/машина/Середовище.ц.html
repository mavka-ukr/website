взяти визначення мавка;

простір мавка {
  дія створити_середовище(
    М: *Машина,
    зовні: *Середовище
  ) -> *Середовище {
    змінна середовище = виділити_середовище(М);

    якщо зовні != пусто {
      середовище.зовні = зовні;
      зовні.кількість_запозичень += 1;
    }

    вернути середовище;
  }

  дія записати_в_середовище(
    М: *Машина,
    середовище: *Середовище,
    назва: *ПредметТексту,
    значення: Значення
  ) {
    призначити(
      М,
      середовище.призначення::адреса,
      назва,
      значення
    );
  }

  дія знайти_в_середовищі(
    М: *Машина,
    середовище: *Середовище,
    назва: *ПредметТексту,
    вихід_значення: *Значення
  ) -> логічне {
    якщо знайти_призначене(
      М,
      середовище.призначення,
      назва,
      вихід_значення
    ) {
      вернути так;
    }

    якщо середовище.зовні != пусто {
      вернути знайти_в_середовищі(
        М,
        середовище.зовні,
        назва,
        вихід_значення
      );
    }

    вернути ні;
  }
}
