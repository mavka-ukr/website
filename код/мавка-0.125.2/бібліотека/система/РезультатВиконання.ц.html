взяти визначення мавка/бібліотека;
взяти визначення мавка/пристрій;

простір мавка {
  простір бібліотека {
    дія записати_клас__система__РезультатВиконання(
      М: *Машина,
      предмет_модуля: *ПредметМодуля
    ) {
      змінна кдбм = М.користувацькі_дані_бібліотеки.значення як *КористувацькіДаніБібліотекиМавки;

      змінна параметр_стдвив = створити_параметр(
        М,
        М.константи_назв._стдвив,
        пусто,
        пусто
      );
      змінна параметр_стдпом = створити_параметр(
        М,
        М.константи_назв._стдпом,
        пусто,
        пусто
      );
      змінна параметр_код = створити_параметр(
        М,
        М.константи_назв._код,
        пусто,
        пусто
      );

      змінна параметри = виділити<Значення>(М, 3);
      параметри[0] = ПРЕДМЕТ(параметр_стдвив);
      параметри[1] = ПРЕДМЕТ(параметр_стдпом);
      параметри[2] = ПРЕДМЕТ(параметр_код);

      змінна послідовність_параметрів = створити_послідовність_без_копіювання(
        М,
        3,
        параметри
      );

      кдбм.система__РезультатВиконання = створити_клас(
        М,
        М.константи_назв._РезультатВиконання,
        послідовність_параметрів
      );

      записати_властивість_модуля(
        М,
        предмет_модуля,
        М.константи_назв._РезультатВиконання,
        ПРЕДМЕТ(кдбм.система__РезультатВиконання)
      );
    }
  }
}
