взяти визначення мавка;

простір мавка {
  дія зробити_накопичувач_тексту(
    М: *Машина,
    вмісткість: природне
  ) -> НакопичувачТексту {
    змінна дані: *п8 = пусто;

    якщо вмісткість > 0 {
      дані = виділити<п8>(М, вмісткість);
    }

    вернути НакопичувачТексту {
      розмір = 0,
      дані = дані,
      вмісткість = вмісткість
    };
  }

  дія покласти_в_накопичувач_тексту(
    М: *Машина,
    накопичувач: *НакопичувачТексту,
    значення: кд
  ) {
    змінна новий_розмір = накопичувач.розмір + значення.розмір;

    накопичувач.дані = перевиділити<п8>(М, накопичувач.дані, новий_розмір);

    змінна п: природне = 0;
    поки п < значення.розмір {
      накопичувач.дані[накопичувач.розмір + п] = значення.дані[п];
      п += 1;
    }

    накопичувач.розмір = новий_розмір;
  }
}
