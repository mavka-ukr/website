взяти визначення мавка;

простір мавка {
  дія створити_і_призначити_модуль_Мавка(М: *Машина) {
    змінна модуль_мавка = створити_модуль(
      М,
      створити_текст(М, "мавка")
    );

    записати_властивість_модуля(
      М,
      модуль_мавка,
      створити_текст(М, "версія"),
      ПРЕДМЕТ(М.версія.предмет_тексту)
    );

    записати_в_середовище(
      М,
      М.глобальне_середовище,
      створити_текст(М, "мавка"),
      ПРЕДМЕТ(модуль_мавка)
    );
  }
}
