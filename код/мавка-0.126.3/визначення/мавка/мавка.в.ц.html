простір мавка {
  внутрішня дія порівняти_тексти(
    М: *Машина,
    а: *ПредметТексту,
    б: *ПредметТексту
  ) -> логічне {
    якщо а == б {
      вернути так;
    }

    якщо а.значення.розмір != б.значення.розмір {
      вернути ні;
    }

    змінна п: природне = 0;
    поки п < а.значення.розмір {
      якщо а.значення.дані[п] != б.значення.дані[п] {
        вернути ні;
      }

      п += 1;
    }

    вернути так;
  }

  внутрішня дія порівняти_послідовності(
    М: *Машина,
    а: *ПредметПослідовності,
    б: *ПредметПослідовності
  ) -> логічне {
    якщо а == б {
      вернути так;
    }

    якщо а.розмір != б.розмір {
      вернути ні;
    }

    змінна п: природне = 0;
    поки п < а.розмір {
      якщо не порівняти_значення(М, а.елементи[п], б.елементи[п]) {
        вернути ні;
      }

      п += 1;
    }

    вернути так;
  }
