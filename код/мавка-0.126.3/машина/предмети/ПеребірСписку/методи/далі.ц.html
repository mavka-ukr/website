взяти визначення мавка;

простір мавка {
  внутрішня дія рідна_дія_метода_структури_ПеребірСписку__далі(
    М: *Машина,
    діяльність: *Діяльність
  ) -> ЗначенняАбоСтанПадіння {
    якщо діяльність.здійснювач.вид != ЗПредмет {
      вернути ПАДІННЯ(М, створити_текст(М, "Неможливо здійснити дію перебору списку \"далі\"."));
    }

    змінна предмет_перебору_списку = діяльність.здійснювач.дані як *Предмет як *ПредметПереборуСписку;

    якщо предмет_перебору_списку.клас != М.клас_ПеребірСписку {
      вернути ПАДІННЯ(М, створити_текст(М, "Неможливо здійснити дію перебору списку \"далі\"."));
    }

    якщо предмет_перебору_списку.почато {
      предмет_перебору_списку.позиція += 1;
    } інакше {
      предмет_перебору_списку.позиція = 0;
      предмет_перебору_списку.почато = так;
    }

    якщо предмет_перебору_списку.позиція < предмет_перебору_списку.предмет_списку.розмір {
      предмет_перебору_списку.поточне_значення = предмет_перебору_списку.предмет_списку.елементи[предмет_перебору_списку.позиція];

      вернути ДІЙСНЕ;
    }

    вернути НЕДІЙСНЕ;
  }

  дія записати_метод_класу_ПеребірСписку__далі(
    М: *Машина
  ) {
    змінна предмет_дії = створити_рідну_дію(
      М,
      М.константи_назв._далі,
      М.пуста_послідовність,
      пусто,
      рідна_дія_метода_структури_ПеребірСписку__далі,
      пусто
    );

    записати_метод_класу(
      М,
      М.клас_ПеребірСписку,
      М.константи_назв._далі,
      предмет_дії
    );
  }
}
