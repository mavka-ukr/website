      М,
      значення,
      значення_властивості,

  // позначка: прохід по типах предметів
  дія отримати_текст(
    М: *Машина,
    значення: Значення,
    місцезнаходження: Місцезнаходження
  ) -> *ПредметТексту {
    якщо значення.вид == ЗНедійсне {
      вернути М.константи_назв._недійсне;
    }

    якщо значення.вид == ЗДійсне {
      вернути М.константи_назв._дійсне;
    }

    якщо значення.вид == ЗЧисло {
      змінна значення_кд: кд;

      якщо перетворити_р64_в_кд(М, значення.дані як р64, значення_кд::адреса) {
        вернути створити_текст_без_копіювання(М, значення_кд.розмір, значення_кд.дані);
      }

      МІСЦЕПАДІННЯ(М, створити_текст(М, "Не вдалось перетворити число в текст."), місцезнаходження);
      вернути пусто;
    }

    змінна предмет = значення.дані як *Предмет;

    якщо предмет.клас == М.клас_Текст {
      вернути предмет як *ПредметТексту;
    }

    змінна результат = здійснити_властивість(
      М,
      значення,
      М.константи_назв._чародія_текст,
      Задіяні { ВидЗадіяних2, 0, пусто, 0 },
      місцезнаходження
    );

    якщо М.стан_падіння {
      вернути пусто;
    }

    якщо результат.вид != ЗПредмет {
      МІСЦЕПАДІННЯ(М, створити_текст(М, "Не вдалось перетворити в текст."), місцезнаходження);
      вернути пусто;
    }

    змінна предмет_результату = результат.дані як *Предмет;

    якщо предмет_результату.клас != М.клас_Текст {
      МІСЦЕПАДІННЯ(М, створити_текст(М, "Не вдалось перетворити в текст."), місцезнаходження);
      вернути пусто;
    }

    вернути предмет_результату як *ПредметТексту;
  }

      МІСЦЕПАДІННЯ(М, створити_текст(М, "Неможливо отримати послідовність з недійсного."), місцезнаходження);
      вернути пусто;
    }

    якщо значення.вид == ЗДійсне {
      МІСЦЕПАДІННЯ(М, створити_текст(М, "Неможливо отримати послідовність із значення \"дійсне\"."), місцезнаходження);
      вернути пусто;
    }

    якщо значення.вид == ЗЧисло {
      МІСЦЕПАДІННЯ(М, створити_текст(М, "Неможливо отримати послідовність з числа."), місцезнаходження);
      вернути пусто;
    }

    змінна предмет = значення.дані як *Предмет;

    якщо предмет.клас == М.клас_Послідовність {
      вернути предмет як *ПредметПослідовності;
    }

    змінна результат = здійснити_властивість(
      М,
      значення,
      М.константи_назв._чародія_послідовність,
      Задіяні { ВидЗадіяних2, 0, пусто, 0 },
      місцезнаходження
    );

    якщо М.стан_падіння {
      вернути пусто;
    }

    якщо результат.вид != ЗПредмет {
      МІСЦЕПАДІННЯ(М, створити_текст(М, "Не вдалось отримати послідовність."), місцезнаходження);
      вернути пусто;
    }

    змінна предмет_результату = результат.дані як *Предмет;

    якщо предмет_результату.клас != М.клас_Послідовність {
      МІСЦЕПАДІННЯ(М, створити_текст(М, "Не вдалось отримати послідовність."), місцезнаходження);
      вернути пусто;
    }

    вернути предмет_результату як *ПредметПослідовності;
  }

  // позначка: прохід по типах предметів
  дія отримати_проміжок(
    М: *Машина,
    від: Значення,
    до: Значення,
    включно: логічне,
    місцезнаходження: Місцезнаходження
  ) -> ЗначенняАбоСтанПадіння {
    якщо від.вид == ЗЧисло {
      якщо до.вид == ЗЧисло {
        вернути ПРЕДМЕТ(створити_проміжок_чисел(
          М,
          від.дані як р64,
          до.дані як р64,
          включно
        ));
      }

      вернути МІСЦЕПАДІННЯ(М, створити_текст(М, "Неможливо створити проміжок (до не є Число)."), місцезнаходження);
    }

    змінна задіяні = Задіяні { ВидЗадіяних2, 2, виділити<Значення>(М, 2), 0 };
    задіяні.дані[0] = до;
    якщо включно {
      задіяні.дані[1] = ДІЙСНЕ;
    } інакше {
      задіяні.дані[1] = НЕДІЙСНЕ;
    }

    вернути здійснити_властивість(
      М,
      від,
      М.константи_назв._чародія_межа,
      задіяні,
      місцезнаходження
    );
  }

  // позначка: прохід по типах предметів
  дія отримати_перебір(
    М: *Машина,
    значення: Значення,
    місцезнаходження: Місцезнаходження
  ) -> ЗначенняАбоСтанПадіння {
    якщо значення.вид == ЗПредмет {
      змінна предмет = значення.дані як *Предмет;

      якщо предмет.клас == М.клас_Дані {
        вернути ПРЕДМЕТ(створити_перебір_даних(
          М,
          предмет як *ПредметДаних
        ));
      }

      якщо предмет.клас == М.клас_ПроміжокЧисел {
        вернути ПРЕДМЕТ(створити_перебір_проміжка_чисел(
          М,
          предмет як *ПредметПроміжкаЧисел
        ));
      }

      якщо предмет.клас == М.клас_Послідовність {
        вернути ПРЕДМЕТ(створити_перебір_послідовності(
          М,
          предмет як *ПредметПослідовності
        ));
      }

      якщо предмет.клас == М.клас_Словник {
        вернути ПРЕДМЕТ(створити_перебір_словника(
          М,
          предмет як *ПредметСловника
        ));
      }

      якщо предмет.клас == М.клас_Список {
        вернути ПРЕДМЕТ(створити_перебір_списку(
          М,
          предмет як *ПредметСписку
        ));
      }

      якщо предмет.клас == М.клас_Текст {
        вернути ПРЕДМЕТ(створити_перебір_тексту(
          М,
          предмет як *ПредметТексту
        ));
      }
    }

    вернути здійснити_властивість(
      М,
      значення,
      М.константи_назв._чародія_перебір,
      Задіяні { ВидЗадіяних2, 0, пусто, 0 },
      місцезнаходження
    );
  }

  внутрішня дія є_невизначеність(значення: р64) -> логічне {
    вернути значення != значення;
  }

  дія порівняти_значення(
    М: *Машина,
    а: Значення,
    б: Значення
  ) -> логічне {
    якщо а.вид != б.вид {
      вернути ні;
    }

    якщо а.вид == ЗНедійсне {
      вернути так;
    }

    якщо а.вид == ЗДійсне {
      вернути так;
    }

    якщо а.вид == ЗЧисло {
      якщо є_невизначеність(а.дані як р64) і є_невизначеність(б.дані як р64) {
        вернути ні;
      }

      вернути (а.дані як р64) == (б.дані як р64);
    }

    якщо а.вид != ЗПредмет {
      ЗАПАНІКУВАТИ(М, "Невідомий вид значення у перевірці на рівність.");
    }

    змінна а_предмет = а.дані як *Предмет;
    змінна б_предмет = б.дані як *Предмет;

    якщо а_предмет == б_предмет {
      вернути так;
    }

    якщо а_предмет.клас == б_предмет.клас {
      якщо а_предмет.клас == М.клас_Текст {
        вернути порівняти_тексти(
          М,
          а_предмет як *ПредметТексту,
          б_предмет як *ПредметТексту
        );
      }

      якщо а_предмет.клас == М.клас_Послідовність {
        вернути порівняти_послідовності(
          М,
          а_предмет як *ПредметПослідовності,
          б_предмет як *ПредметПослідовності
        );
      }
    }

    вернути ні;
  }

  дія перевірити_чи_предмет_є(
    М: *Машина,
    а: *Предмет,
    б: *ПредметКласу
  ) -> логічне {
    вернути а.клас == б;
  }
}
