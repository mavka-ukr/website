  простір бібліотека {

        значення = "ПОМИЛКА_АРГУМЕНТУ";

        значення = "Адреса використовується.";

      } інакше якщо помилка == МАВКА_ІНЕТ_ПОМИЛКА_ЗʼЄДНАННЯ_СКИНОТО {
        значення = "Зʼєднання скинуто.";

        значення = "Хост недоступний.";
      } інакше якщо помилка == МАВКА_ІНЕТ_ПОМИЛКА_ТАЙМАУТ {
        значення = "Час вийшов.";
      } інакше якщо помилка == МАВКА_ІНЕТ_ПОМИЛКА_КАНАЛ_ЗЛАМАНО {
        значення = "Канал зламано.";
      } інакше якщо помилка == МАВКА_ІНЕТ_ПОМИЛКА_ДОСТУПУ {
        значення = "Помилка доступу.";

        значення = "Адреса недоступна.";

        значення = "Переповнення буфера.";

        значення = "Мережа відмовила.";

    дія виділити_аргумент_інет_звʼязку(М: *Машина) -> *АргументІнетЗвʼязку {
      змінна аз = виділити<АргументІнетЗвʼязку>(М, 1);

      аз.М = М;
      аз.прт_звʼязок = пусто;
      аз.як_текст = ні;
      аз.прт_обробник_підключення = пусто;
      аз.прт_обробник_даних = пусто;
      аз.прт_обробник_стікання = пусто;
      аз.прт_обробник_закінчення = пусто;
      аз.прт_обробник_помилки = пусто;
      аз.прт_обробник_відключення = пусто;

      вернути аз;
    }

    дія обробник_знищення_інет_звʼязку(аргумент_звʼязку: адреса) {
      змінна аз = аргумент_звʼязку як *АргументІнетЗвʼязку;
      змінна М = аз.М;

      якщо аз.прт_звʼязок != пусто {
        змінна предмет_звʼязку = аз.прт_звʼязок.предмет як *СкладенийПредмет;
        змінна кдз = предмет_звʼязку.користувацькі_дані.значення як *КористувацькіДаніІнетЗвʼязку;

        кдз.адр_звʼязок = пусто;

        відпустити_притриманий_глобально(М, аз.прт_звʼязок);
      }
      якщо аз.прт_обробник_підключення != пусто {
        відпустити_притриманий_глобально(М, аз.прт_обробник_підключення);
      }
      якщо аз.прт_обробник_даних != пусто {
        відпустити_притриманий_глобально(М, аз.прт_обробник_даних);
      }
      якщо аз.прт_обробник_стікання != пусто {
        відпустити_притриманий_глобально(М, аз.прт_обробник_стікання);
      }
      якщо аз.прт_обробник_закінчення != пусто {
        відпустити_притриманий_глобально(М, аз.прт_обробник_закінчення);
      }
      якщо аз.прт_обробник_помилки != пусто {
        відпустити_притриманий_глобально(М, аз.прт_обробник_помилки);
      }
      якщо аз.прт_обробник_відключення != пусто {
        відпустити_притриманий_глобально(М, аз.прт_обробник_відключення);
      }

      звільнити(М, аз);
    }

    дія створити_інет_Звʼязок(М: *Машина, адр_звʼязок: адреса) -> *СкладенийПредмет {
      змінна кдбм = М.користувацькі_дані_бібліотеки.значення як *КористувацькіДаніБібліотекиМавки;

      змінна предмет_звʼязку = створити_складений_предмет(М, кдбм.інет__Звʼязок);

      змінна кдз = виділити<КористувацькіДаніІнетЗвʼязку>(М, 1);

      кдз.адр_звʼязок = адр_звʼязок;

      предмет_звʼязку.користувацькі_дані = створити_користувацькі_дані(М, кдз, перед_знищенням_користувацьких_даних_інет_звʼязку);

      вернути предмет_звʼязку;
    }

    дія перед_знищенням_користувацьких_даних_інет_звʼязку(М: *Машина, користувацькі_дані: *КористувацькіДані) {
      змінна кдз = користувацькі_дані.значення як *КористувацькіДаніІнетЗвʼязку;

        надрукувати_кд(М, "ПОМИЛКА МАВКИ: кдз.адр_звʼязок != пусто!!!");
      }
      звільнити(М, кдз);
    }

      М: *Машина,

      записати_дію__інет__служити(М, предмет_модуля);
      записати_дію__інет__підключитись(М, предмет_модуля);
    }
