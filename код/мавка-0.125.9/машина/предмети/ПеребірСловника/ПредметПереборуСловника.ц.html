взяти визначення мавка;

простір мавка {
  дія записати_метод_класу_ПеребірСловника__чародія_текст(М: *Машина);
  дія записати_метод_класу_ПеребірСловника__далі(М: *Машина);

  внутрішня дія записати_ознаки_класу_ПеребірСловника(М: *Машина) {
    змінна кількість_ознак = природне(2);
    змінна дані_ознак = виділити<Значення>(М, кількість_ознак);

    дані_ознак[0] = ПРЕДМЕТ(створити_ознаку(
      М,
      М.константи_назв._значення,
      пусто,
      НЕДІЙСНЕ
    ));
    дані_ознак[1] = ПРЕДМЕТ(створити_ознаку(
      М,
      М.константи_назв._словник,
      пусто,
      НЕДІЙСНЕ
    ));

    М.клас_ПеребірСловника.ознаки = створити_послідовність_без_копіювання(
      М,
      кількість_ознак,
      дані_ознак
    );
  }

  внутрішня дія записати_властивість_класу_Словник__Перебір(М: *Машина) {
    клас_змінити_властивість(
      М,
      М.клас_Словник,
      М.константи_назв._Перебір,
      ПРЕДМЕТ(М.клас_ПеребірСловника),
      Місцезнаходження { пусто, 0 }
    );

    якщо М.стан_падіння {
      ЗАПАНІКУВАТИ(М, "Помилка при записі властивості \"Перебір\" для класу \"Словник\".");
    }
  }

  дія заповнити_клас_ПеребірСловника(
    М: *Машина
  ) {
    М.клас_ПеребірСловника.вбудований = так;
    М.клас_ПеребірСловника.заборонити_наслідування = так;

    записати_ознаки_класу_ПеребірСловника(М);

    записати_метод_класу_ПеребірСловника__чародія_текст(М);
    записати_метод_класу_ПеребірСловника__далі(М);

    записати_властивість_класу_Словник__Перебір(М);
  }

  дія створити_перебір_словника(
    М: *Машина,
    предмет_словника: *ПредметСловника
  ) -> *ПредметПереборуСловника {
    змінна предмет = виділити_предмет<ПредметПереборуСловника>(М, М.клас_ПеребірСловника);

    предмет.предмет_словника = предмет_словника;
    предмет.почато = ні;
    предмет.позиція = 0;
    предмет.поточне_значення = НЕДІЙСНЕ;

    вернути предмет;
  }

  дія здійснити_клас_перебору_словника(
    М: *Машина,
    задіяні: Задіяні,
    місцезнаходження: Місцезнаходження
  ) -> ЗначенняАбоСтанПадіння {
    вернути МІСЦЕПАДІННЯ(М, створити_текст(М, "Неможливо здійснити клас \"ПеребірСловника\"."), місцезнаходження);
  }

  дія предмет_перебору_словника_отримати_властивість(
    М: *Машина,
    предмет_перебору_словника: *ПредметПереборуСловника,
    назва: *ПредметТексту,
    місцезнаходження: Місцезнаходження
  ) -> ЗначенняАбоСтанПадіння {
    якщо порівняти_тексти(М, назва, М.константи_назв._значення) {
      вернути предмет_перебору_словника.поточне_значення;
    }
    якщо порівняти_тексти(М, назва, М.константи_назв._словник) {
      вернути ПРЕДМЕТ(предмет_перебору_словника.предмет_словника);
    }

    змінна предмет_методу = знайти_метод_в_класі(
      М,
      М.клас_ПеребірСловника,
      назва
    );

    якщо предмет_методу == пусто {
      вернути НЕДІЙСНЕ;
    }

    вернути ПРЕДМЕТ(предмет_методу);
  }
}
